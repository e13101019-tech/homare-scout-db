/** * スカウトDB API (GAS Web App) * ------------------------------------------------ * フロント(GitHub Pages)からfetchでアクセスするJSON API。 * スプレッドシート1つ(Master)をバックエンドDBとして使う。 * * Masterシート列構成 (1行目ヘッダー): * A:timestamp B:year C:playerId D:rank E:staff F:team G:name * H:school I:posMain J:posBest K:posSub L:gpa M:pitchHand N:batHand * O:feature P:lodging Q:scholarship R:otherNotes S:decision * * decision: "入学予定" | "保留／決定待ち" | "なし" */ const SHEET_ID = '1_Ee6w-4gOqflK_HNgAAhGLmA7FstiTgRe_9IfkIaAwE'; const SHEET_NAME = 'Master'; const HEADERS = [ 'timestamp', 'year', 'playerId', 'rank', 'staff', 'team', 'name', 'school', 'posMain', 'posBest', 'posSub', 'gpa', 'pitchHand', 'batHand', 'feature', 'lodging', 'scholarship', 'otherNotes', 'decision', 'entryId' ]; // 編集で上書き可能なフィールド(timestamp/playerId/entryIdは編集不可) const EDITABLE_FIELDS = [ 'year', 'rank', 'staff', 'team', 'name', 'school', 'posMain', 'posBest', 'posSub', 'gpa', 'pitchHand', 'batHand', 'feature', 'lodging', 'scholarship', 'otherNotes', 'decision' ]; function getSheet_() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); } return sheet; } function readAllRecords_() { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); if (values.length < 2) return []; const rows = values.slice(1); return rows.map(row => { const rec = {}; HEADERS.forEach((h, i) => { rec[h] = row[i]; }); return rec; }).filter(rec => rec.name); // 空行除外 } function jsonOut_(obj) { // GAS Web Appは「全員(匿名可)」でデプロイすると自動でAccess-Control-Allow-Origin: *が付与される。 // フロント側はfetchのContent-Typeを"text/plain;charset=utf-8"にしてpreflightを回避すること。 return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } const REQUIRED_FIELDS = ['year', 'rank', 'staff', 'team', 'name', 'decision']; const DECISION_VALUES = ['入学予定', '保留／決定待ち', 'なし']; const RANK_VALUES = ['1', '2', '3']; /** * 入力バリデーション。エラーがあれば文字列配列を返す。空配列ならOK。 */ function validateSubmit_(body) { const errors = []; REQUIRED_FIELDS.forEach(f => { if (body[f] === undefined || String(body[f]).trim() === '') { errors.push(`必須項目が未入力です: ${f}`); } }); if (body.decision !== undefined && !DECISION_VALUES.includes(body.decision)) { errors.push(`decisionの値が不正です: ${body.decision}`); } if (body.rank !== undefined && !RANK_VALUES.includes(String(body.rank))) { errors.push(`rankの値が不正です: ${body.rank}`); } if (body.year !== undefined && !/^\d{4}$/.test(String(body.year))) { errors.push(`yearは4桁の年で指定してください: ${body.year}`); } if (body.mode === 'existing' && !body.playerId) { errors.push('modeがexistingの場合はplayerIdが必須です'); } return errors; } /** * 文字列項目はtrimしてから保存する */ function sanitizeBody_(body) { const cleaned = {}; Object.keys(body).forEach(k => { cleaned[k] = typeof body[k] === 'string' ? body[k].trim() : body[k]; }); return cleaned; } /** * GET * ?action=list&year=2029 → 年度の全レコード * ?action=dashboard&year=2029 → 決定件数の集計 * ?action=years → 登録済み年度の一覧 */ function doGet(e) { try { const action = e.parameter.action; if (action === 'years') { const records = readAllRecords_(); const years = [...new Set(records.map(r => r.year))].sort(); return jsonOut_({ ok: true, years }); } if (action === 'list') { const year = e.parameter.year; if (!year) return jsonOut_({ ok: false, error: 'yearパラメータが必要です' }); const records = readAllRecords_().filter(r => String(r.year) === String(year)); // ソート指定に対応: ?sort=rank&order=asc const sortKey = e.parameter.sort; const order = e.parameter.order === 'desc' ? -1 : 1; if (sortKey && HEADERS.includes(sortKey)) { records.sort((a, b) => { if (a[sortKey] < b[sortKey]) return -1 * order; if (a[sortKey] > b[sortKey]) return 1 * order; return 0; }); } return jsonOut_({ ok: true, records }); } if (action === 'dashboard') { const year = e.parameter.year; if (!year) return jsonOut_({ ok: false, error: 'yearパラメータが必要です' }); const records = readAllRecords_().filter(r => String(r.year) === String(year)); const summary = { total: records.length, '入学予定': 0, '保留／決定待ち': 0, 'なし': 0 }; records.forEach(r => { if (summary.hasOwnProperty(r.decision)) summary[r.decision]++; }); return jsonOut_({ ok: true, summary }); } return jsonOut_({ ok: false, error: 'unknown action' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } /** * POST (JSON body) * action: "checkDuplicate" { name, team } * → { ok:true, match: {playerId,...} | null } * * action: "submit" { mode: "new"|"existing", playerId(existing時), ...fields } * → { ok:true, playerId } */ function doPost(e) { try { const raw = JSON.parse(e.postData.contents); const body = sanitizeBody_(raw); const action = body.action; if (action === 'checkDuplicate') { if (!body.name || !body.team) { return jsonOut_({ ok: false, error: 'nameとteamが必要です' }); } const records = readAllRecords_(); const match = records.find(r => String(r.name).trim() === String(body.name).trim() && String(r.team).trim() === String(body.team).trim() ); return jsonOut_({ ok: true, match: match || null }); } if (action === 'submit') { const errors = validateSubmit_(body); if (errors.length > 0) { return jsonOut_({ ok: false, error: 'validation', details: errors }); } // LockServiceで同時書き込みによる採番衝突を防ぐ const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sheet = getSheet_(); const playerId = body.mode === 'existing' && body.playerId ? body.playerId : generatePlayerId_(); const entryId = generateEntryId_(); const row = HEADERS.map(h => { if (h === 'timestamp') return new Date(); if (h === 'playerId') return playerId; if (h === 'entryId') return entryId; return body[h] !== undefined ? body[h] : ''; }); sheet.appendRow(row); return jsonOut_({ ok: true, playerId, entryId }); } finally { lock.releaseLock(); } } if (action === 'update') { if (!body.entryId) { return jsonOut_({ ok: false, error: 'entryIdが必要です' }); } const errors = validateSubmit_(body); if (errors.length > 0) { return jsonOut_({ ok: false, error: 'validation', details: errors }); } const lock2 = LockService.getScriptLock(); lock2.waitLock(10000); try { const sheet = getSheet_(); const rowIndex = findRowIndexByEntryId_(sheet, body.entryId); if (rowIndex === -1) { return jsonOut_({ ok: false, error: '対象のレコードが見つかりません' }); } // EDITABLE_FIELDSのみ上書き。timestamp/playerId/entryIdは維持 EDITABLE_FIELDS.forEach(field => { const colIndex = HEADERS.indexOf(field) + 1; // 1-indexed if (body[field] !== undefined) { sheet.getRange(rowIndex, colIndex).setValue(body[field]); } }); return jsonOut_({ ok: true, entryId: body.entryId }); } finally { lock2.releaseLock(); } } return jsonOut_({ ok: false, error: 'unknown action' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } function findRowIndexByEntryId_(sheet, entryId) { const values = sheet.getDataRange().getValues(); const entryIdCol = HEADERS.indexOf('entryId'); for (let i = 1; i < values.length; i++) { if (String(values[i][entryIdCol]) === String(entryId)) { return i + 1; // 1-indexed row number } } return -1; } function generatePlayerId_() { // 簡易連番ID: P + タイムスタンプ下6桁 + ランダム2桁 return 'P' + Utilities.formatDate(new Date(), 'JST', 'yyMMddHHmmss').slice(-8); } function generateEntryId_() { // 各レコードを一意に識別するID(ミリ秒まで含めて衝突回避) return 'E' + Utilities.formatDate(new Date(), 'JST', 'yyMMddHHmmssSSS'); } /** * 【一度だけ実行】entryId列が空の既存行にIDを振る移行スクリプト。 * GASエディタの関数選択プルダウンから backfillEntryIds を選んで実行してください。 */ function backfillEntryIds() { const sheet = getSheet_(); // ヘッダー行にentryId列が無ければ追加 const headerRow = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const entryIdCol = HEADERS.indexOf('entryId') + 1; // 1-indexed if (headerRow[entryIdCol - 1] !== 'entryId') { sheet.getRange(1, entryIdCol).setValue('entryId'); } const values = sheet.getDataRange().getValues(); let count = 0; for (let i = 1; i < values.length; i++) { const row = values[i]; if (!row[0]) continue; // 空行スキップ(timestampが空) const currentEntryId = row[entryIdCol - 1]; if (!currentEntryId) { sheet.getRange(i + 1, entryIdCol).setValue(generateEntryId_()); Utilities.sleep(5); // ミリ秒IDの衝突を避けるための微小待機 count++; } } Logger.log(`entryIdを ${count} 件のレコードに付与しました`); } /** * エディタ上の「実行」ボタンでdoGetをテストするための関数。 * 実行後、実行ログにJSON結果が表示される。 */ function test_doGet_years() { const result = doGet({ parameter: { action: 'years' } }); Logger.log(result.getContent()); } function test_doGet_list() { const result = doGet({ parameter: { action: 'list', year: '2029' } }); Logger.log(result.getContent()); } function test_doGet_dashboard() { const result = doGet({ parameter: { action: 'dashboard', year: '2029' } });
